	g: string, -- procedural garment name (Logic/CharacterLook.GARMENTS)
	catalog: number?, -- Roblox catalog accessory id; 0 / nil = procedural
	asset: string?, -- key of ClassLooks.assets (owner's model); replaces the procedural garment when the model exists
	role: string?, -- colour role of the garment (default: the garment's own default role in GARMENTS)
}
export type OutfitSpec = {
	colors: { main: string, second: string, trim: string?, leather: string?, metal: string? },
	body: { [string]: string }?, -- R15 part name -> colour role ("main", "second", "trim", "leather", "metal", "skin"); default BODY_ROLES
	garments: { Garment },
	shirt: number?, -- Roblox classic Shirt id (0 = the procedural body colours show)
	pants: number?, -- Roblox classic Pants id
}
export type ClassLook = {
	class: string,
	nameKey: string, -- "classlook.<Class>.name" (the default outfit's name in the wardrobe)
	outfit: OutfitSpec,
	-- Later (not in DESIGN-17): armour from the equipment changes the look by slot, as in Margonem.
	layers: { chest: OutfitSpec?, head: OutfitSpec?, legs: OutfitSpec?, hands: OutfitSpec?, feet: OutfitSpec?, back: OutfitSpec? },
}
export type Face = { id: string, catalog: number?, eyes: string, brows: string, mouth: string, mark: string? }
export type Skin = { id: string, color: string }
export type Hair = { id: string, catalog: number?, asset: string?, style: string, color: string }
export type Asset = { model: string, attach: string?, offset: { number }?, scale: number? }

### C. Czysta logika (`src/shared/Logic/CharacterLook.luau`, nowy; `Logic/Premium.luau`; schemat)

**C1. Słowniki i walidacja** (bez API Roblox):
- `CharacterLook.GARMENTS: { [string]: { slot: string, pieces: number, role: string, hidesHair: boolean? } }`: każdy procedural (dzisiejsze `hood`, `crown`, `halo`, `horns`, `mask`, `cape`, `pauldrons` + nowe z A2, A3 i B2, w tym style włosów `hair_short`, `hair_long`, `hair_ponytail`, `hair_braid`, `hair_topknot`, `hair_curly`, `hair_swept`), `slot` z `{ "head", "face", "back", "chest", "shoulders", "arms", "hands", "waist", "legs", "feet" }` (te same nazwy, które kiedyś posłużą warstwom zbroi), `pieces` = liczba części, które rysuje `AvatarBuild`, domyślna rola koloru, `hidesHair = true` dla nakryć głowy (`hood`, `helm_crest`, `crown` nie). Komentarz: „Util/AvatarBuild draws every name here; tests/classlooks.spec checks every look uses only these.”
- `CharacterLook.FACE_PARTS = { eyes = { … }, brows = { … }, mouth = { … }, mark = { … } }` (A3).
- `CharacterLook.normalize(look: any): { face: string, skin: string, hair: string }` — każde pole istniejące w `ClassLooks`, inaczej to z `DEFAULT_LOOK`; nie-tabela → kopia `DEFAULT_LOOK`. Nigdy nie zwraca wejścia (nowa tabela).
- `CharacterLook.isValid(face, skin, hair): (boolean, string?)` — dla remote'ów: typy `string` i istnienie każdego id, błąd `look.error.bad`.

**C2. Rozwiązanie wyglądu.**
- `CharacterLook.outfitSpec(def: Cosmetics.Def): OutfitSpec` — `def.outfit` albo z dzisiejszych pól: `colors = { main = color, second = color2 or color }`, `garments` z `parts` (każda nazwa → `{ g = name }`), `body` = role `BODY_MAIN` → `main`, `BODY_SECOND` → `second`, dłonie `skin` (dziś `CosmeticLook` maluje też dłonie na `second`: zostaw to w trybie awatara, D3, i opisz w DECYZJE, jeśli zmieniasz).
- `CharacterLook.allowed(def, classId): boolean` — `def.kind == "outfit"` i (`def.class == nil` albo `def.class == classId`).
- `CharacterLook.outfitOf(account, classId): string?` — `account.cosmetics.classOutfits[classId]`, jeśli istnieje, jest posiadany i `allowed`; inaczej `nil` (= domyślny strój klasy).
- `CharacterLook.resolve(input): LookSpec` — wejście `{ classId, look, outfitId: string?, dye: any?, robloxAvatar: boolean, lure: boolean? }`, wyjście:

```lua
export type LookSpec = {
	mode: "class" | "avatar",
	skin: string?, -- hex; nil in avatar mode (the avatar keeps its own)
	face: { catalog: number, eyes: string, brows: string, mouth: string, mark: string? }?,
	hair: { catalog: number, asset: string?, style: string, color: string }?, -- nil = shaved or avatar mode
	body: { [string]: string }, -- R15 part -> hex (empty in avatar mode without an outfit)
	garments: { { g: string, color: string, catalog: number, asset: string? } }, -- procedural or catalog / asset pieces, hats hide hair
	shirt: number, pants: number, -- catalog clothing (0 = none)
	accessories: { number }, -- non-zero catalog ids (garments, hair, in that order)
	catalogKey: string, -- stable text of every catalog id (face, shirt, pants, accessories): the server re-applies the description only when it changes
	pieces: number, -- procedural parts drawn (Config.CharacterLook.maxPieces)
}
```

  Zasady: w trybie `class` strój = `outfitSpec(Cosmetics.get(outfitId))`, gdy `outfitId` dany i `allowed`, inaczej `ClassLooks.get(classId).outfit`; barwnik S123 (`dye` = id kosmetyku z `equipped.dye`; z S123, sprawdź w kodzie i w sekcji SESJE) przemalowuje role `main` i `second` stroju **w danych spec**, tą samą regułą co S123: `CosmeticRules.dyed({ color = colors.main, color2 = colors.second } :: any, Cosmetics.get(dye))` (S123: `dyed(outfit: Def?, dye: Def?): (string?, string?)`, `nil, nil` bez barwnika); działa także na strój domyślny klasy (w trybie klasy strój jest zawsze), a w trybie awatara tylko z założonym strojem (DECYZJE); `Logic/CharacterLook` wymaga `Logic/CosmeticRules` (sprawdź, że `CosmeticRules` nie wymaga `Logic/Premium`, inaczej cykl); skóra z `look.skin` dla ról `skin`; fryzura z `look.hair`, chyba że któryś element stroju ma `hidesHair`; peleryna stroju (`cape`, slot `back`) **zostaje w spec** także przy `lure == true`: pierwszeństwo peleryny przynęty S122 to umowa na modelu, nie w danych (D1: część peleryny nazywa się `OutfitCape` i jest przezroczysta, gdy model ma atrybut `LureCape == true`; `CosmeticLook.applyLure(model, on)` z S122 przełącza ją bez przebudowy stroju), więc `lure` w wejściu służy tylko do liczenia `pieces` (budżet z peleryną przynęty); element z `catalog ~= 0` idzie do `accessories` (fryzura z katalogu tak samo), z `catalog == 0` i `asset` do rysowania modelem (zapas: procedural, gdy brak pliku, D1), inaczej procedural. W trybie `avatar` (`robloxAvatar = true`): `face`, `hair`, `skin` = `nil`, a strój (jeśli założony) nakłada się jak dziś: role `body` i procedurale z `outfitSpec`, `shirt`/`pants` z katalogu podmieniają ubranie awatara; bez założonego stroju `body` i `garments` puste. `pieces` = suma `GARMENTS[g].pieces` rysowanych proceduralnie (plus fryzura).
- Wynik `resolve` jest jedynym źródłem kolorów stroju: po `AvatarBuild.applyProcedural` serwer i podgląd **nie** wołają już `CosmeticLook.applyDye` (S123) na tym modelu, bo przemalowałby części drugi raz wg starych `BODY_MAIN` / `BODY_SECOND`; `applyDye` zostaje jako funkcja zgodności (ścieżka `applyOutfit`, D1).
- `CharacterLook.questOutfits(questId, classId): { string }` i `CharacterLook.bossWinOutfits(bossId, classId): { string }` — id strojów z `obtain`, posortowane.
